// Pseudocode: Online Exam Portal (OOPs)
Class User
    Attributes:
        username
        password
    Method login():
        // base login logic

Class Student extends User
    Method login():
        If student is registered AND enrolled in course
            Allow access to exams
        Else
            Deny access

Class Admin extends User
    Method login():
        If credentials valid AND user has admin privileges
            Allow access to exam management tools
        Else
            Deny access

// Example Run
Create Student s with username "john123", password "pass"
If s.login() succeeds
    Show available exams
Create Admin a
If a.login() succeeds
    Show exam management tools
