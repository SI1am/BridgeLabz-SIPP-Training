// Pseudocode: Attendance Marker (2D Array)
Initialize 2D array attendance[30][30] // 30 students, 30 days
// Mark attendance
attendance[studentIndex][dayIndex] = 1 // Present
attendance[studentIndex][dayIndex] = 0 // Absent
// Daily summary
For day in 0 to 29
    count = 0
    For student in 0 to 29
        If attendance[student][day] == 1
            count++
    Print "Day", day+1, ":", count, "students present"
// Monthly summary
For student in 0 to 29
    total = 0
    For day in 0 to 29
        If attendance[student][day] == 1
            total++
    Print "Student", student+1, ":", total, "days present"
// Defaulter list (<75%)
For student in 0 to 29
    total = count of 1s in attendance[student]
    If total < 22.5
        Print "Student", student+1, "is a defaulter"
