// Pseudocode: Online Survey Report Generator (Collectors.groupingBy, Aggregation)
Class SurveyResponse
    Attributes:
        restaurantName
        rating (1-5)
Input: List<SurveyResponse> responses
// Group by restaurant and average
Map<String, Double> avgRatings = responses.stream()
    .collect(groupingBy(restaurantName, averaging(rating)))
// Find highest and lowest
maxRestaurant = restaurant with max avgRatings
minRestaurant = restaurant with min avgRatings
Print avgRatings
Print maxRestaurant, minRestaurant
